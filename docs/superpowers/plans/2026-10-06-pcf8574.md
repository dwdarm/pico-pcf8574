# PCF8574 Driver Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a full-featured C PCF8574 I2C GPIO-expander driver for RP2040 with host tests, examples, and docs.

**Architecture:** Cached synchronous driver — `pcf8574_t` holds `i2c_inst_t*`, address, and output-latch cache; all pin writes are read-modify-write on cache to respect quasi-bidirectional I/O; host tests shadow `<hardware/i2c.h>` with a mock.

**Tech Stack:** C11, Raspberry Pi pico-sdk (`pico_stdlib`, `hardware_i2c`), gcc host tests, CMake 3.13+

## Global Constraints

- Language is C with pico-sdk, no C++ or MicroPython bindings.
- Driver uses only `<hardware/i2c.h>`, `<stdint.h>`, `<stdbool.h>`, `<stddef.h>` — no malloc, no panic, no asserts.
- Valid addresses are 0x20-0x27 (PCF8574) and 0x38-0x3F (PCF8574A).
- I2C timeout define is `#define PCF8574_I2C_TIMEOUT_US 10000`, overridable at compile time.
- All public functions return `bool`, true on ACK success, false on invalid arg or I2C NACK/timeout.
- Caller owns `i2c_init`, SDA/SCL gpio_set_function, and baudrate (100 kHz default, 400 kHz supported).
- I2C reads never happen inside IRQ — examples poll INT pin in main loop.

// pcf8574.h — PCF8574/PCF8574A I2C GPIO expander driver for RP2040 (pico-sdk)
#ifndef PCF8574_H
#define PCF8574_H
#include <stdint.h>
#include <stdbool.h>
#include "hardware/i2c.h"

#ifndef PCF8574_I2C_TIMEOUT_US
#define PCF8574_I2C_TIMEOUT_US 10000u
#endif

#define PCF8574_ADDR_MIN 0x20u
#define PCF8574_ADDR_MAX 0x27u
#define PCF8574A_ADDR_MIN 0x38u
#define PCF8574A_ADDR_MAX 0x3Fu

typedef struct {
    i2c_inst_t *i2c;
    uint8_t addr;
    uint8_t out_state;
} pcf8574_t;

static inline bool pcf8574_addr_valid(uint8_t addr) {
    return ((addr >= PCF8574_ADDR_MIN && addr <= PCF8574_ADDR_MAX) ||
            (addr >= PCF8574A_ADDR_MIN && addr <= PCF8574A_ADDR_MAX));
}

bool pcf8574_init(pcf8574_t *dev, i2c_inst_t *i2c, uint8_t addr, uint8_t initial);
bool pcf8574_write_byte(pcf8574_t *dev, uint8_t val);
bool pcf8574_read_byte(pcf8574_t *dev, uint8_t *val);
bool pcf8574_write_pin(pcf8574_t *dev, uint8_t pin, bool level);
bool pcf8574_read_pin(pcf8574_t *dev, uint8_t pin, bool *level);
bool pcf8574_toggle_pin(pcf8574_t *dev, uint8_t pin);
bool pcf8574_set_inputs_mask(pcf8574_t *dev, uint8_t mask);
bool pcf8574_is_connected(pcf8574_t *dev);
#endif

#include "pcf8574.h"
#include <stddef.h>

bool pcf8574_init(pcf8574_t *dev, i2c_inst_t *i2c, uint8_t addr, uint8_t initial) {
    if (!dev || !i2c) return false;
    if (!pcf8574_addr_valid(addr)) return false;
    dev->i2c = i2c;
    dev->addr = addr;
    dev->out_state = initial;
    int rc = i2c_write_blocking(i2c, addr, &initial, 1, false);
    if (rc != 1) return false;
    return true;
}

bool pcf8574_write_byte(pcf8574_t *dev, uint8_t val) {
    if (!dev || !dev->i2c) return false;
    int rc = i2c_write_blocking(dev->i2c, dev->addr, &val, 1, false);
    if (rc != 1) return false;
    dev->out_state = val;
    return true;
}

bool pcf8574_read_byte(pcf8574_t *dev, uint8_t *val) {
    if (!dev || !dev->i2c || !val) return false;
    uint8_t b = 0;
    int rc = i2c_read_blocking(dev->i2c, dev->addr, &b, 1, false);
    if (rc != 1) return false;
    *val = b;
    return true;
}

// Stubs to satisfy header link — full impl in Task 2
bool pcf8574_write_pin(pcf8574_t *dev, uint8_t pin, bool level) { (void)dev; (void)pin; (void)level; return false; }
bool pcf8574_read_pin(pcf8574_t *dev, uint8_t pin, bool *level) { (void)dev; (void)pin; (void)level; return false; }
bool pcf8574_toggle_pin(pcf8574_t *dev, uint8_t pin) { (void)dev; (void)pin; return false; }
bool pcf8574_set_inputs_mask(pcf8574_t *dev, uint8_t mask) { (void)dev; (void)mask; return false; }
bool pcf8574_is_connected(pcf8574_t *dev) { (void)dev; return false; }

bool pcf8574_write_pin(pcf8574_t *dev, uint8_t pin, bool level) {
    if (!dev || !dev->i2c) return false;
    if (pin > 7) return false;
    uint8_t next = level ? (dev->out_state | (uint8_t)(1u << pin))
                         : (dev->out_state & (uint8_t)~(1u << pin));
    int rc = i2c_write_blocking(dev->i2c, dev->addr, &next, 1, false);
    if (rc != 1) return false;
    dev->out_state = next;
    return true;
}

bool pcf8574_read_pin(pcf8574_t *dev, uint8_t pin, bool *level) {
    if (!dev || !dev->i2c || !level) return false;
    if (pin > 7) return false;
    uint8_t b = 0;
    int rc = i2c_read_blocking(dev->i2c, dev->addr, &b, 1, false);
    if (rc != 1) return false;
    *level = (b & (uint8_t)(1u << pin)) != 0;
    return true;
}

bool pcf8574_toggle_pin(pcf8574_t *dev, uint8_t pin) {
    if (!dev || !dev->i2c) return false;
    if (pin > 7) return false;
    uint8_t next = dev->out_state ^ (uint8_t)(1u << pin);
    int rc = i2c_write_blocking(dev->i2c, dev->addr, &next, 1, false);
    if (rc != 1) return false;
    dev->out_state = next;
    return true;
}

bool pcf8574_set_inputs_mask(pcf8574_t *dev, uint8_t mask) {
    if (!dev || !dev->i2c) return false;
    uint8_t next = dev->out_state | mask;
    int rc = i2c_write_blocking(dev->i2c, dev->addr, &next, 1, false);
    if (rc != 1) return false;
    dev->out_state = next;
    return true;
}

bool pcf8574_is_connected(pcf8574_t *dev) {
    if (!dev || !dev->i2c) return false;
    uint8_t b = 0;
    int rc = i2c_read_blocking(dev->i2c, dev->addr, &b, 1, false);
    return rc == 1;
}

// input_buttons — upper nibble as inputs with INT poll + optional IRQ flag
#include <stdio.h>
#include "pico/stdlib.h"
#include "hardware/i2c.h"
#include "hardware/gpio.h"
#include "pcf8574.h"

#define INT_PIN 22
static volatile bool int_fired = false;
static void int_cb(uint gpio, uint32_t events) {
    (void)gpio; (void)events;
    int_fired = true; // never do I2C in IRQ
}

int main(void) {
    stdio_init_all();
    i2c_init(i2c0, 100 * 1000);
    gpio_set_function(4, GPIO_FUNC_I2C);
    gpio_set_function(5, GPIO_FUNC_I2C);
    gpio_pull_up(4); gpio_pull_up(5);
    gpio_init(INT_PIN); gpio_set_dir(INT_PIN, GPIO_IN); gpio_pull_up(INT_PIN);
    gpio_set_irq_enabled_with_callback(INT_PIN, GPIO_IRQ_EDGE_FALL, true, &int_cb);
    pcf8574_t dev;
    if (!pcf8574_init(&dev, i2c0, 0x20, 0xFF)) { printf("pcf8574 not found\n"); return 1; }
    pcf8574_set_inputs_mask(&dev, 0xF0); // P4-P7 inputs, P0-P3 outputs
    uint8_t last = 0xFF;
    while (true) {
        if (!gpio_get(INT_PIN)) int_fired = true; // poll fallback if IRQ missed
        if (int_fired) {
            int_fired = false;
            uint8_t v = 0;
            if (pcf8574_read_byte(&dev, &v) && v != last) {
                last = v;
                printf("inputs: 0x%02X\n", v);
            }
        }
        sleep_ms(10);
    }
}
```

# pico-pcf8574

C driver for PCF8574 / PCF8574A I2C 8-bit GPIO expander on RP2040 (pico-sdk).

## Wiring

| PCF8574 | Pico |
|---|---|
| SDA | GP4 (I2C0 SDA) |
| SCL | GP5 (I2C0 SCL) |
| A0-A2 | GND/VCC per address |
| INT | GP22 (pull-up) + GND common |

Addresses: PCF8574 `0x20-0x27`, PCF8574A `0x38-0x3F`. 100 kHz default, 400 kHz supported.

## Usage

```c
i2c_init(i2c0, 100*1000);
gpio_set_function(4, GPIO_FUNC_I2C);
gpio_set_function(5, GPIO_FUNC_I2C);
gpio_pull_up(4); gpio_pull_up(5);
pcf8574_t dev;
pcf8574_init(&dev, i2c0, 0x20, 0xFF);
pcf8574_set_inputs_mask(&dev, 0xF0);
pcf8574_write_pin(&dev, 0, false);
uint8_t v; pcf8574_read_byte(&dev, &v);
```

Inputs must be written `1` (quasi-bidirectional). Driver keeps `1`s in cache for input pins.

## Build

```
export PICO_SDK_PATH=/path/to/pico-sdk
cmake -B build -DPICO_SDK_PATH=$PICO_SDK_PATH
cmake --build build
```
